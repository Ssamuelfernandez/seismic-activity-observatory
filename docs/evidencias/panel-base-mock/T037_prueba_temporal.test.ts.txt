import { createElement } from 'react';
import { renderToStaticMarkup } from 'react-dom/server';
import { expect, it } from 'vitest';
import { Attribution } from '../src/components/Attribution';

const TEXT = 'Sismos: USGS · Fronteras: Natural Earth (dominio público) · Globo: react-globe.gl y three (MIT) · Sin textura';
const lum = (hex: string) => {
  const c = [1, 3, 5].map((i) => parseInt(hex.slice(i, i + 2), 16) / 255).map((v) => (v <= 0.03928 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4));
  return 0.2126 * c[0] + 0.7152 * c[1] + 0.0722 * c[2];
};
const ratio = (a: string, b: string) => { const [x, y] = [lum(a), lum(b)].sort((p, q) => q - p); return (x + 0.05) / (y + 0.05); };

it('atribución', () => {
  for (const variant of ['desktop', 'mobile'] as const) {
    const html = renderToStaticMarkup(createElement(Attribution, { variant }));
    console.log(`${variant}: ${html}`);
    expect(html.replace(/<[^>]+>/g, '')).toBe(TEXT);
    expect(html).toContain('text-[11px]');
    expect(html).not.toMatch(/text-\[(9|10)px\]|text-xs/);
  }
  console.log(`contraste #9ca3af sobre #000000 (hud-bg): ${ratio('#9ca3af', '#000000').toFixed(2)} : 1`);
  console.log(`contraste #9ca3af sobre #0b0d10 (hud-panel): ${ratio('#9ca3af', '#0b0d10').toFixed(2)} : 1`);
  expect(ratio('#9ca3af', '#0b0d10')).toBeGreaterThanOrEqual(4.5);
});
